    84→    84→
    85→    85→export function Importar() {
    86→    86→  const { triggerRefresh, setActiveTab, activeProjectId } = useAppStore();
    87→    87→  const { toast } = useToast();
    88→    88→  const fileRef = useRef<HTMLInputElement>(null);
    89→    89→
    90→    90→  const [dragOver, setDragOver] = useState(false);
    91→    91→  const [loading, setLoading] = useState(false);
    92→    92→  const [importing, setImporting] = useState(false);
    93→    93→  const [paso, setPaso] = useState<Paso>("subir");
    94→    94→  const [result, setResult] = useState<ImportResult | null>(null);
    95→    95→  const [selected, setSelected] = useState<Set<number>>(new Set());
    96→    96→  const [expanded, setExpanded] = useState<Set<number>>(new Set());
    97→    97→  const [selectedAll, setSelectedAll] = useState(false);
    98→    98→  const [importResult, setImportResult] = useState<{
    99→    99→    importadas: number;
   100→   100→    omitidasFecha: number;
   101→   101→    errores: string[];
   102→   102→    resumen: string[];
   103→   103→  } | null>(null);
   104→   104→
   105→   105→  // Números que ya existen en la BD (solo informativo)
   106→   106→  const [numerosExistentes, setNumerosExistentes] = useState<Set<string>>(new Set());
   107→   107→
   108→   108→  // Archivo subido para "Ver en Excel"
   109→   109→  const [uploadedFile, setUploadedFile] = useState<File | null>(null);
   110→   110→
   111→   111→  // Dialog para ver datos del Excel
   112→   112→  const [viewingSheet, setViewingSheet] = useState<{
   113→   113→    hoja: string;
   114→   114→    datos: CeldaRaw[][];
   115→   115→  } | null>(null);
   116→   116→  const [loadingSheet, setLoadingSheet] = useState(false);
   117→   117→
   118→   118→  // Cuando el resultado del análisis llega, verificar cuáles ya existen
   119→   119→  useEffect(() => {
   120→   120→    if (!result || !activeProjectId) return;
   121→   121→
   122→   122→    const numeros = result.facturas
   123→   123→      .filter((f) => f.exito && f.numero)
   124→   124→      .map((f) => f.numero);
   125→   125→
   126→   126→    if (numeros.length === 0) return;
   127→   127→
   128→   128→    // Consultar API de existentes (solo informativo)
   129→   129→    fetch(`/api/import/existing?projectId=${activeProjectId}&numeros=${encodeURIComponent(numeros.join(","))}`)
   130→   130→      .then((r) => r.json())
   131→   131→      .then((data: { existentes: ExistingInvoice[] }) => {
   132→   132→        const set = new Set<string>(data.existentes.map((e) => e.numero));
   133→   133→        setNumerosExistentes(set);
   134→   134→      })
   135→   135→      .catch(() => {});
   136→   136→  }, [result, activeProjectId]);
   137→   137→
   138→   138→  const procesarArchivo = useCallback(async (file: File) => {
   139→   139→    setLoading(true);
   140→   140→    setResult(null);
   141→   141→    setNumerosExistentes(new Set());
   142→   142→    setPaso("subir");
   143→   143→    setUploadedFile(file);
   144→   144→
   145→   145→    try {
   146→   146→      const fd = new FormData();
   147→   147→      fd.append("archivo", file);
   148→   148→
   149→   149→      const res = await fetch("/api/import", { method: "POST", body: fd });
   150→   150→      const data = await res.json() as ImportResult;
   151→   151→      if (!res.ok) throw new Error((data as any).error);
   152→   152→
   153→   153→      setResult(data);
   154→   154→
   155→   155→      // NO auto-seleccionar. El usuario revisa y decide.
   156→   156→      setSelected(new Set());
   157→   157→      setSelectedAll(false);
   158→   158→      setPaso("revision");
   159→   159→    } catch (err) {
   160→   160→      toast({
   161→   161→        title: "Error al procesar",
   162→   162→        description: err instanceof Error ? err.message : "No se pudo leer el archivo.",
   163→   163→        variant: "destructive",
   164→   164→      });
   165→   165→    } finally {
   166→   166→      setLoading(false);
   167→   167→    }
   168→   168→  }, [toast]);
   169→   169→
   170→   170→  // Ver datos del Excel para una hoja específica
   171→   171→  const verEnExcel = useCallback(async (sheetName: string) => {
   172→   172→    if (!uploadedFile) {
   173→   173→      toast({ title: "Archivo no disponible", description: "El archivo original ya no está en memoria.", variant: "destructive" });
   174→   174→      return;
   175→   175→    }
   176→   176→
   177→   177→    setLoadingSheet(true);
   178→   178→    try {
   179→   179→      const buffer = await uploadedFile.arrayBuffer();
   180→   180→      const XLSX = await import("xlsx");
   181→   181→      const wb = XLSX.read(buffer, { type: "array" });
   182→   182→      const ws = wb.Sheets[sheetName];
   183→   183→      if (!ws || !ws["!ref"]) {
   184→   184→        toast({ title: "Hoja vacía", description: `La hoja "${sheetName}" no tiene datos.` });
   185→   185→        setLoadingSheet(false);
   186→   186→        return;
   187→   187→      }
   188→   188→
   189→   189→      const range = XLSX.utils.decode_range(ws["!ref"]);
   190→   190→      const datos: CeldaRaw[][] = [];
   191→   191→
   192→   192→      for (let r = range.s.r; r <= range.e.r; r++) {
   193→   193→        const row: CeldaRaw[] = [];
   194→   194→        let hasData = false;
   195→   195→        for (let c = range.s.c; c <= range.e.c; c++) {
   196→   196→          const addr = XLSX.utils.encode_cell({ r, c });
   197→   197→          const cell = ws[addr];
   198→   198→          const valor = cell?.v !== undefined ? cell.v : null;
   199→   199→          const coord = XLSX.utils.encode_cell({ r, c });
   200→   200→          row.push({ coord, valor });
   201→   201→          if (valor !== null && valor !== "") hasData = true;
   202→   202→        }
   203→   203→        if (hasData) datos.push(row);
   204→   204→      }
   205→   205→
   206→   206→      setViewingSheet({ hoja: sheetName, datos });
   207→   207→    } catch {
   208→   208→      toast({ title: "Error", description: "No se pudo leer la hoja del Excel.", variant: "destructive" });
   209→   209→    } finally {
   210→   210→      setLoadingSheet(false);
   211→   211→    }
   212→   212→  }, [uploadedFile, toast]);
   213→   213→
   214→   214→  const handleDrop = useCallback((e: React.DragEvent) => {
   215→   215→    e.preventDefault(); setDragOver(false);
   216→   216→    const file = e.dataTransfer.files[0];
   217→   217→    if (file) procesarArchivo(file);
   218→   218→  }, [procesarArchivo]);
   219→   219→
   220→   220→  const handleFileInput = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
   221→   221→    const file = e.target.files?.[0];
   222→   222→    if (file) procesarArchivo(file);
   223→   223→  }, [procesarArchivo]);
   224→   224→
   225→   225→  const confirmarImport = async () => {
   226→   226→    if (!result) return;
   227→   227→    const facturasAImportar = result.facturas.filter((_, i) => selected.has(i));
   228→   228→    if (facturasAImportar.length === 0) {
   229→   229→      toast({ title: "Nada que importar", description: "Seleccioná al menos una factura.", variant: "destructive" });
   230→   230→      return;
   231→   231→    }
   232→   232→
   233→   233→    setImporting(true);
   234→   234→    try {
   235→   235→      const res = await fetch("/api/import/confirm", {
   236→   236→        method: "POST",
   237→   237→        headers: { "Content-Type": "application/json" },
   238→   238→        body: JSON.stringify({ facturas: facturasAImportar, projectId: activeProjectId }),
   239→   239→      });
   240→   240→      const data = await res.json();
   241→   241→      if (!res.ok) throw new Error(data.error);
   242→   242→
   243→   243→      setImportResult({
   244→   244→        importadas: data.importadas,
   245→   245→        omitidasFecha: data.omitidasFecha || 0,
   246→   246→        errores: data.errores || [],
   247→   247→        resumen: data.resumen || [],
   248→   248→      });
   249→   249→      setPaso("resultado");
   250→   250→      triggerRefresh();
   251→   251→    } catch (err) {
   252→   252→      toast({ title: "Error al guardar", description: err instanceof Error ? err.message : "Intentá de nuevo.", variant: "destructive" });
   253→   253→    } finally {
   254→   254→      setImporting(false);
   255→   255→    }
   256→   256→  };
   257→   257→
   258→   258→  const toggleSelect = (idx: number) => {
   259→   259→    setSelected((prev) => {
   260→   260→      const next = new Set(prev);
   261→   261→      if (next.has(idx)) next.delete(idx); else next.add(idx);
   262→   262→      setSelectedAll(false);
   263→   263→      return next;
   264→   264→    });
   265→   265→  };
   266→   266→
   267→   267→  const toggleAll = () => {
   268→   268→    if (!result) return;
   269→   269→    if (selectedAll) {
   270→   270→      setSelected(new Set()); setSelectedAll(false);
   271→   271→    } else {
   272→   272→      // Seleccionar TODAS las exitosas
   273→   273→      const all = new Set<number>();
   274→   274→      result.facturas.forEach((f, i) => { if (f.exito) all.add(i); });
   275→   275→      setSelected(all); setSelectedAll(true);
   276→   276→    }
   277→   277→  };
   278→   278→
   279→   279→  const toggleExpand = (idx: number) => {
   280→   280→    setExpanded((prev) => { const next = new Set(prev); if (next.has(idx)) next.delete(idx); else next.add(idx); return next; });
   281→   281→  };
   282→   282→
   283→   283→  const exitosasSeleccionadas = result ? result.facturas.filter((_, i) => selected.has(i) && result.facturas[i].exito).length : 0;
   284→   284→  const conAdvertencias = result ? result.facturas.filter((f) => f.advertencias && f.advertencias.length > 0).length : 0;
   285→   285→  const sinFecha = result ? result.facturas.filter((f) => f._fechaDudosa || !f.fecha).length : 0;
   286→   286→  const yaExisten = result
   287→   287→    ? result.facturas.filter((f) => f.numero && numerosExistentes.has(f.numero)).length
   288→   288→    : 0;
   289→   289→  const conPrecioCero = result
   290→   290→    ? result.facturas.filter((f) => f.productos.some((p) => p._precioCero)).length
   291→   291→    : 0;
   292→   292→  const conDiscrepancia = result
   293→   293→    ? result.facturas.filter((f) => f._totalDiscrepancia).length
   294→   294→    : 0;
   295→   295→
   296→   296→  // PASO: SUBIR ARCHIVO
   297→   297→  if (paso === "subir") {
   298→   298→    return (
   299→   299→      <div className="space-y-6 animate-in fade-in max-w-4xl">
   300→   300→        <div>
   301→   301→          <h2 className="text-xl font-bold text-stone-900">Importar facturas</h2>
   302→   302→          <p className="text-sm text-stone-500 mt-0.5">Subí el archivo Excel con las hojas de factura</p>
   303→   303→        </div>
   304→   304→
   305→   305→        <Card className={`border-2 border-dashed transition-all cursor-pointer ${dragOver ? "border-purple-500 bg-purple-50" : "border-stone-300 hover:border-purple-400 hover:bg-stone-50"}`}
   306→   306→          onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
   307→   307→          onDragLeave={() => setDragOver(false)}
   308→   308→          onDrop={handleDrop}
   309→   309→          onClick={() => fileRef.current?.click()}
   310→   310→        >
   311→   311→          <CardContent className="p-10 text-center">
   312→   312→            {loading ? (
   313→   313→              <div className="space-y-3">
   314→   314→                <Loader2 className="w-10 h-10 mx-auto text-purple-700 animate-spin" />
   315→   315→                <p className="text-stone-600 font-medium">Leyendo hojas del archivo...</p>
   316→   316→                <p className="text-xs text-stone-400">Esto puede tardar unos segundos con muchas hojas</p>
   317→   317→              </div>
   318→   318→            ) : (
   319→   319→              <div className="space-y-3">
   320→   320→                <div className="w-14 h-14 mx-auto rounded-2xl bg-purple-100 flex items-center justify-center"><Upload className="w-7 h-7 text-purple-700" /></div>
   321→   321→                <p className="text-stone-900 font-semibold">Arrastrá el archivo Excel aquí</p>
   322→   322→                <p className="text-sm text-stone-500">o hacé clic para seleccionarlo</p>
   323→   323→                <p className="text-xs text-stone-400">Acepta archivos .xlsx con una o más hojas de factura</p>
   324→   324→              </div>
   325→   325→            )}
   326→   326→          </CardContent>
   327→   327→          <input ref={fileRef} type="file" accept=".xlsx,.xls" onChange={handleFileInput} className="hidden" />
   328→   328→        </Card>
   329→   329→
   330→   330→        <div className="flex items-start gap-3 p-4 rounded-xl bg-sky-50 border border-sky-200">
   331→   331→          <FileSpreadsheet className="w-5 h-5 text-sky-600 flex-shrink-0 mt-0.5" />
   332→   332→          <p className="text-sm text-sky-800"><strong>Formato esperado:</strong> Cada hoja del Excel debe ser una factura con las columnas CANTIDAD, CODIGO, DESCRIPCION, P.UNI y TOTAL. El sistema detecta automáticamente el formato.</p>
   333→   333→        </div>
   334→   334→
   335→   335→        <div className="flex items-start gap-3 p-4 rounded-xl bg-emerald-50 border border-emerald-200">
   336→   336→          <ShieldCheck className="w-5 h-5 text-emerald-600 flex-shrink-0 mt-0.5" />
   337→   337→          <p className="text-sm text-emerald-800"><strong>Vos revisás y decidís:</strong> El sistema lee los datos tal cual están. No se elimina ni se modifica nada automáticamente. Podés verificar cada factura contra el Excel original antes de confirmar. Un cliente puede comprar varias veces — eso no es duplicado.</p>
   338→   338→        </div>
   339→   339→      </div>
   340→   340→    );
   341→   341→  }
   342→   342→
   343→   343→  // PASO: RESULTADO FINAL
   344→   344→  if (paso === "resultado" && importResult) {
   345→   345→    return (
   346→   346→      <div className="space-y-6 animate-in fade-in max-w-4xl">
   347→   347→        <div>
   348→   348→          <h2 className="text-xl font-bold text-stone-900">Importación completada</h2>
   349→   349→          <p className="text-sm text-stone-500 mt-0.5">Resultado de la importación</p>
   350→   350→        </div>
   351→   351→
   352→   352→        <Card className="border-0 shadow-sm">
   353→   353→          <CardContent className="p-8 text-center">
   354→   354→            <div className="w-16 h-16 mx-auto mb-4 rounded-full bg-emerald-100 flex items-center justify-center">
   355→   355→              <CheckCircle2 className="w-8 h-8 text-emerald-700" />
   356→   356→            </div>
   357→   357→            <h3 className="text-lg font-bold text-stone-900 mb-1">Importación completada</h3>
   358→   358→            <p className="text-3xl font-bold text-purple-700 my-3">{importResult.importadas} factura(s) importada(s)</p>
   359→   359→            <div className="flex justify-center gap-6 text-sm text-stone-500">
   360→   360→              {importResult.omitidasFecha > 0 && <span className="text-rose-600">{importResult.omitidasFecha} rechazada(s) por falta de fecha</span>}
   361→   361→            </div>
   362→   362→          </CardContent>
   363→   363→        </Card>
   364→   364→
   365→   365→        {importResult.resumen && importResult.resumen.length > 0 && (
   366→   366→          <Card className="border border-sky-200">
   367→   367→            <CardContent className="p-5">
   368→   368→              <h4 className="font-semibold text-sky-700 mb-2">Resumen</h4>
   369→   369→              <ul className="space-y-1">
   370→   370→                {importResult.resumen.map((r, i) => <li key={i} className="text-xs text-sky-600">- {r}</li>)}
   371→   371→              </ul>
   372→   372→            </CardContent>
   373→   373→          </Card>
   374→   374→        )}
   375→   375→
   376→   376→        {importResult.errores.length > 0 && (
   377→   377→          <Card className="border border-rose-200">
   378→   378→            <CardContent className="p-5">
   379→   379→              <h4 className="font-semibold text-rose-700 mb-2">Errores detallados</h4>
   380→   380→              <ul className="space-y-1 max-h-40 overflow-y-auto">
   381→   381→                {importResult.errores.map((err, i) => <li key={i} className="text-xs text-rose-600 font-mono">- {err}</li>)}
   382→   382→              </ul>
   383→   383→            </CardContent>
   384→   384→          </Card>
   385→   385→        )}
   386→   386→
   387→   387→        <div className="flex gap-3">
   388→   388→          <Button variant="outline" onClick={() => { setPaso("subir"); setResult(null); setImportResult(null); setNumerosExistentes(new Set()); setUploadedFile(null); if (fileRef.current) fileRef.current.value = ""; }}>
   389→   389→            Importar otro archivo
   390→   390→          </Button>
   391→   391→          <Button onClick={() => setActiveTab("consultar")} className="bg-purple-800 hover:bg-purple-900 text-white font-semibold shadow-md shadow-purple-200">
   392→   392→            Ver facturas importadas
   393→   393→          </Button>
   394→   394→        </div>
   395→   395→      </div>
   396→   396→    );
   397→   397→  }
   398→   398→
   399→   399→  // PASO: REVISIÓN
   400→   400→  if (!result) return null;
   401→   401→
   402→   402→  return (
   403→   403→    <>
   404→   404→      <div className="space-y-5 animate-in fade-in max-w-4xl">
   405→   405→        <div className="flex items-center justify-between">
   406→   406→          <div>
   407→   407→            <h2 className="text-xl font-bold text-stone-900">Revisión de datos</h2>
   408→   408→            <p className="text-sm text-stone-500 mt-0.5">Revisá cada factura. Seleccioná las que querés importar.</p>
   409→   409→          </div>
   410→   410→          <Button variant="ghost" size="sm" onClick={() => { setPaso("subir"); setResult(null); setNumerosExistentes(new Set()); setUploadedFile(null); if (fileRef.current) fileRef.current.value = ""; }}>
   411→   411→            Volver
   412→   412→          </Button>
   413→   413→        </div>
   414→   414→
   415→   415→        {/* Barra de progreso */}
   416→   416→        <div className="flex items-center gap-2 text-xs">
   417→   417→          <div className="flex items-center gap-1.5"><ShieldCheck className="w-4 h-4 text-purple-700" /><span className="font-medium text-purple-800">Paso 1: Cargar archivo</span></div>
